const  LEFT_KEY = 0b00000010;
const RIGHT_KEY = 0b00000100;
const  DOWN_KEY = 0b00001000;
const     Z_KEY = 0b00010000;
const     X_KEY = 0b00100000;

fn
sleep(tm)
{
  var  base = sys.timer();

    loop
    {
        if (sys.timer()-base) >= tm
        {
          break;
        }


      halt;
    }
}


fn
dot2(x,y,pixel)
{
  offscreen_address[VIDEO_W*y][x] = pixel;
}


fn
dot(x,y,pixel)
{
  (offscreen_address+(4*VIDEO_W*y)+(4*x)).u32ref = pixel;
}


fn
draw_rect(x,y,w,h,pixel)
{
    for off in w
    {
      dot(x+off,y    ,pixel);
      dot(x+off,y+h-1,pixel);
    }


    for off in h
    {
      dot(x    ,y+off,pixel);
      dot(x+w-1,y+off,pixel);
    }
}


fn
fill_rect(x,y,w,h,pixel)
{
    for y_off in h{
    for x_off in w{
      dot(x+x_off,y+y_off,pixel);
    }}
}


fn
print_int(i,f,x,y,w,pixel)
{
    if i == 0
    {
      f('0',x,y,pixel);

      return;
    }


    while i
    {
      f('0'+(i%10),x,y,pixel);

      x -=  w;
      i /= 10;
    }
}


fn
print8_unicode(u,x,y,pixel)
{
  var  ptr = FONT8_START+(8*u);

    for y_off in 8
    {
      var  bits = ptr.u8ref;
                  ptr += 1;

        for x_off in 8
        {
            if bits&0x80
            {
              dot(x+x_off,y+y_off,pixel);
            }


          bits <<= 1;
        }
    }
}


fn
print14_unicode(u,x,y,pixel)
{
  var  ptr = FONT14_START+(2*14*u);

    for y_off in 14
    {
      var  bits = ptr.u16ref;
                  ptr += 2;

        for x_off in 14
        {
            if bits&0x8000
            {
              dot(x+x_off,y+y_off,pixel);
            }


          bits <<= 1;
        }
    }
}


fn
print14_unicode_s(s,x,y,pixel)
{
    loop
    {
      var  u = s.u16ref;
               s += 2;

        if u == 0
        {
          break;
        }


      print14_unicode(u,x,y,pixel);

      x += 16;
    }
}


fn
fill(x,y,w,h,pixel)
{
  var  base_ptr = offscreen_address+(4*VIDEO_W*y)+(4*x);

    for y_off in h
    {
      var  ptr = base_ptr             ;
                 base_ptr += 4*VIDEO_W;

        for x_off in w
        {
          ptr.u32ref = pixel;

          ptr += 4;
        }
    }
}

MIDDLE_SIZE = BASE_SIZE*BASE_SIZE;

const
LARGE_SIZE = MIDDLE_SIZE*BASE_SIZE;

fn
draw_bmp(ptr,sz,x,y,pix_sz)
{
    for y_off in sz{
    for x_off in sz{
      var  pix = get(ptr,sz,x_off,y_off);

      fill_rect(x+(pix_sz*x_off),y+(pix_sz*y_off),pix_sz,pix_sz,pix);
    }}
}


fn
fill_bmp(ptr,sz,pix)
{
    for y in sz{
    for x in sz{
      ptr.u32ref[(sz*y)+x] = pix;
    }}
}


fn
get(ptr,sz,x,y)
{
  return ptr.u32ref[(sz*y)+x];
}


fn
check_zero(x,y)
{
    if (x >= 0)
    && (y >= 0)
    && (x <  BASE_SIZE)
    && (y <  BASE_SIZE)
    {
      return table[(BASE_SIZE*y)+x] == 0;
    }


  return false;
}


fn
process_table()
{
    for y in BASE_SIZE{
    for x in BASE_SIZE{
        if check_zero(x,y) == 0
        {
          var  v  = check_zero(x-1,y  );
               v |= check_zero(x+1,y  );
               v |= check_zero(x  ,y-1);
               v |= check_zero(x  ,y+1);

            if v
            {
              table[(BASE_SIZE*y)+x] = 2;
            }
        }
    }}
}


fn
table_to_base()
{
  process_table();

    for y in BASE_SIZE{
    for x in BASE_SIZE{
      var  v = 0;

        if table[(BASE_SIZE*y)+x]
        {
          v = 0x00FF0000;
        }


      base[(BASE_SIZE*y)+x] = v;
    }}
}


fn
transfer(dst_ptr,dst_sz,src_ptr,src_sz)
{
    for y in BASE_SIZE{
    for x in BASE_SIZE{
      var  tabval = table[(BASE_SIZE*y)+x];

        if tabval
        {
            for yy in src_sz{
            for xx in src_sz{
              var  v = get(src_ptr,src_sz,xx,yy);

                if (v == 0) && (tabval == 1)
                {
                  v = 0x7FFF7F00;
                }


              dst_ptr.u32ref[(dst_sz*(src_sz*y+yy))+(src_sz*x+xx)] = v;
            }}
        }
    }}
}


fn
large_to_middle()
{
    for y in MIDDLE_SIZE{
    for x in MIDDLE_SIZE{
      var  v = 0;

        for yy in MIDDLE_SIZE{
        for xx in MIDDLE_SIZE{
          v += get(large.ptr,LARGE_SIZE,MIDDLE_SIZE*x+xx,
                                        MIDDLE_SIZE*y+yy);
        }}


      middle[(MIDDLE_SIZE*y)+x] = v(MIDDLE_SIZE*MIDDLE_SIZE);
    }}
}


const
PIXEL_SIZE = 8;

fn
update()
{
  fill(0,0,SCREEN_W,SCREEN_H,0x0000FFFF);

  draw_bmp(base.ptr,BASE_SIZE,0,0,PIXEL_SIZE);

  draw_rect(PIXEL_SIZE*cursor_x,PIXEL_SIZE*cursor_y,PIXEL_SIZE-1,PIXEL_SIZE-1,0x00000000);
  draw_rect(PIXEL_SIZE*cursor_x,PIXEL_SIZE*cursor_y,PIXEL_SIZE  ,PIXEL_SIZE  ,0xFFFFFFFF);

  var  x = PIXEL_SIZE*BASE_SIZE;

//  draw_bmp(middle.ptr,MIDDLE_SIZE,x,0,4);
  draw_bmp(middle.ptr,MIDDLE_SIZE,x,0,2);

  x += MIDDLE_SIZE*2;

  draw_bmp(large.ptr,LARGE_SIZE,x,0,1);

  flip();
}


static cursor_x;
static cursor_y;


fn
flip()
{
  var  tmp = offscreen_address                       ;
             offscreen_address = screen_address      ;
                                 screen_address = tmp;
}


fn
main()
{
     screen_address = video_field.ptr             ;
  offscreen_address = video_field.ptr+(4*SCREEN_W);

  table_to_base();

  transfer(middle.ptr,MIDDLE_SIZE,base.ptr,BASE_SIZE);
  transfer(large.ptr,LARGE_SIZE,middle.ptr,MIDDLE_SIZE);

  update();

    loop
    {
      var  input = sys.input();

        if input&Z_KEY
        {
          var  v = table[(BASE_SIZE*cursor_y)+cursor_x];

          table[(BASE_SIZE*cursor_y)+cursor_x] = !v;

          table_to_base();

          fill_bmp(middle.ptr,MIDDLE_SIZE,0);
          fill_bmp( large.ptr, LARGE_SIZE,0);

          transfer(middle.ptr,MIDDLE_SIZE,base.ptr,BASE_SIZE);
          transfer(large.ptr,LARGE_SIZE,middle.ptr,MIDDLE_SIZE);
        }


           if (input& LEFT_KEY) && (cursor_x >             0){cursor_x -= 1;}
      else if (input&RIGHT_KEY) && (cursor_x < (BASE_SIZE-1)){cursor_x += 1;}

           if (input&  UP_KEY) && (cursor_y >             0){cursor_y -= 1;}
      else if (input&DOWN_KEY) && (cursor_y < (BASE_SIZE-1)){cursor_y += 1;}

        if input
        {
          update();
        }


      halt;
    }
}
