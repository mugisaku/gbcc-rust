static video_field[VIDEO_W*VIDEO_H]: u32;

const VIDEO_W = 400;
const VIDEO_H = 200;

const    UP_KEY = 0b00000001;
const  LEFT_KEY = 0b00000010;
const RIGHT_KEY = 0b00000100;
const  DOWN_KEY = 0b00001000;
const     Z_KEY = 0b00010000;
const     X_KEY = 0b00100000;
const     C_KEY = 0b01000000;
const     V_KEY = 0b10000000;

const FRONT_DIR = 0;
const  LEFT_DIR = 1;
const RIGHT_DIR = 2;
const  BACK_DIR = 3;

const  CHR_W = 24;
const  CHR_H = 40;




fn
sleep(tm)
{
  var  base = sys.timer();

    loop
    {
        if (sys.timer()-base) >= tm
        {
          break;
        }


      halt;
    }
}


fn
dot2(x,y,pixel)
{
  video_field[VIDEO_W*y][x] = pixel;
}


fn
dot(x,y,pixel)
{
  (video_field.ptr+(4*VIDEO_W*y)+(4*x)).u32ref = pixel;
}


fn
draw_rect(x,y,w,h,pixel)
{
    for off in w
    {
      dot(x+off,y    ,pixel);
      dot(x+off,y+h-1,pixel);
    }


    for off in h
    {
      dot(x    ,y+off,pixel);
      dot(x+w-1,y+off,pixel);
    }
}


fn
print_int(i,f,x,y,w,pixel)
{
    if i == 0
    {
      f('0',x,y,pixel);

      return;
    }


    while i
    {
      f('0'+(i%10),x,y,pixel);

      x -=  w;
      i /= 10;
    }
}


fn
print8_unicode(u,x,y,pixel)
{
  var  ptr = FONT8_START+(8*u);

    for y_off in 8
    {
      var  bits = ptr.u8ref;
                  ptr += 1;

        for x_off in 8
        {
            if bits&0x80
            {
              dot(x+x_off,y+y_off,pixel);
            }


          bits <<= 1;
        }
    }
}


fn
print14_unicode(u,x,y,pixel)
{
  var  ptr = FONT14_START+(2*14*u);

    for y_off in 14
    {
      var  bits = ptr.u16ref;
                  ptr += 2;

        for x_off in 14
        {
            if bits&0x8000
            {
              dot(x+x_off,y+y_off,pixel);
            }


          bits <<= 1;
        }
    }
}


fn
print14_unicode_s(s,x,y,pixel)
{
    loop
    {
      var  u = s.u16ref;
               s += 2;

        if u == 0
        {
          break;
        }


      print14_unicode(u,x,y,pixel);

      x += 16;
    }
}


fn
print_character(dir,anim,dst_x,dst_y)
{
  var  src_x = 0;
  var  src_y = 0;
  var  rev = 0;

       if (dir == LEFT_DIR) {src_y = CHR_H;}
  else if (dir == RIGHT_DIR){src_y = CHR_H;  rev = 1;}
  else if (dir == BACK_DIR) {src_y = CHR_H*2;}

       if anim == 1{src_x = CHR_W;}
  else if anim == 3{src_x = CHR_W*2;}

  var  dst_ptr_base = video_field.ptr+(4*VIDEO_W*dst_y)+(4*dst_x);

  var  src_pitch = 4*image[0];
  var  src_ptr_base = image.ptr+8+(src_pitch*src_y)+(4*src_x);

    for y in CHR_H
    {
      var  dst_ptr = dst_ptr_base;
                     dst_ptr_base += 4*VIDEO_W;

      var  src_ptr = src_ptr_base;
                     src_ptr_base += src_pitch;

        if rev
        {
          dst_ptr += 4*CHR_W;

            for x in CHR_W
            {
              dst_ptr.u32ref = src_ptr.u32ref;

              dst_ptr -= 4;
              src_ptr += 4;
            }
        }

      else
        {
            for x in CHR_W
            {
              dst_ptr.u32ref = src_ptr.u32ref;

              dst_ptr += 4;
              src_ptr += 4;
            }
        }
    }
}


fn
fill(x,y,w,h,pixel)
{
  var  base_ptr = video_field.ptr+(4*VIDEO_W*y)+(4*x);

    for y_off in h
    {
      var  ptr = base_ptr             ;
                 base_ptr += 4*VIDEO_W;

        for x_off in w
        {
          ptr.u32ref = pixel;

          ptr += 4;
        }
    }
}


fn
test(x,y)
{
    while x < (VIDEO_W-32)
    {
      draw_rect(x,y,32,32,0xFFFFFFFF);

      x += 8;

      halt;
    }
}


static  g_dir = 0;

static  g_x = 128;
static  g_y = 128;
static  g_anim = 0;
static  g_x_move_value = 0;
static  g_y_move_value = 0;


fn
video_proc()
{
  loop
  {
    fill(0,0,VIDEO_W,VIDEO_H,0);

    print_int(g_x,print14_unicode,16*8, 0,16,0xFFFFFFFF);
    print_int(g_y,print14_unicode,16*8,16,16,0xFFFFFFFF);

    print_character(g_dir,g_anim>>1&3,g_x,g_y);

    halt;
  }
}


fn
object_proc()
{
  loop
  {
    var  input = sys.input();

      if input&Z_KEY
      {
        sys.spawn(test,g_x,g_y);
      }


      if (g_x_move_value == 0) && (g_y_move_value == 0)
      {
             if (input&UP_KEY   ){g_dir =  BACK_DIR;  g_y_move_value = 24;}
        else if (input&DOWN_KEY ){g_dir = FRONT_DIR;  g_y_move_value = 24;}
             if (input&LEFT_KEY ){g_dir =  LEFT_DIR;  g_x_move_value = 24;}
        else if (input&RIGHT_KEY){g_dir = RIGHT_DIR;  g_x_move_value = 24;}
      }

    else
      {
          for _ in 2
          {
                 if g_dir ==  BACK_DIR{g_y_move_value -= 2;  g_y -= 2;}
            else if g_dir == FRONT_DIR{g_y_move_value -= 2;  g_y += 2;}
                 if g_dir ==  LEFT_DIR{g_x_move_value -= 2;  g_x -= 2;}
            else if g_dir == RIGHT_DIR{g_x_move_value -= 2;  g_x += 2;}

            g_anim += 1;
          }
      }


    halt;
  }
}


fn
main()
{
  video = video_field.ptr;

  sys.spawn(video_proc);
  sys.spawn(object_proc);
}
